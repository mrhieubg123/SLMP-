/* Tạo bảng error catalog */
CREATE TABLE ERROR_CATALOG ( ERROR_ID NUMBER GENERATED BY DEFAULT AS IDENTITY, ERROR_CODE VARCHAR2(50 CHAR) NOT NULL, ERROR_NAME NVARCHAR2(255) NOT NULL, ERROR_LEVEL NUMBER(2), DESCRIPTION NVARCHAR2(500), CONSTRAINT PK_ERROR_CATALOG PRIMARY KEY (ERROR_ID), CONSTRAINT UK_ERROR_CATALOG_CODE UNIQUE (ERROR_CODE) );

/* =============
   2. THÔNG TIN MÁY + TRẠNG THÁI REALTIME
   ============= */
CREATE TABLE FATP_MACHINE_INFO
(
    MACHINE_ID          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    MACHINE_CODE        VARCHAR2(50 CHAR) NOT NULL,
    MACHINE_NAME        NVARCHAR2(100),
FACTORY             VARCHAR2(30 CHAR) NOT NULL,
LINE                VARCHAR2(30 CHAR) NOT NULL,

MACHINE_TYPE        VARCHAR2(50 CHAR),
CATEGORY            VARCHAR2(50 CHAR),

LOCATION            VARCHAR2(50 CHAR),
LOCATION_ORDER      NUMBER(10),

IP_ADDRESS          VARCHAR2(50 CHAR),

/* Trạng thái hiện tại */
CURRENT_STATUS      NUMBER(3) DEFAULT 0 NOT NULL,
CURRENT_ERROR_ID    NUMBER,

/* Lần cuối collector nhận được dữ liệu từ máy */
LAST_UPDATE         TIMESTAMP(3),

ENABLED             NUMBER(1) DEFAULT 1 NOT NULL,
CREATED_AT          TIMESTAMP(3) DEFAULT SYSTIMESTAMP NOT NULL,
START_TIME_CURRENT_STATUS   TIMESTAMP(6),

CONSTRAINT PK_FATP_MACHINE_INFO
    PRIMARY KEY (MACHINE_ID),

CONSTRAINT UK_MACHINE_CODE
    UNIQUE (MACHINE_CODE),

CONSTRAINT FK_MACHINE_CURRENT_ERROR
    FOREIGN KEY (CURRENT_ERROR_ID)
    REFERENCES ERROR_CATALOG(ERROR_ID),

CONSTRAINT CK_MACHINE_ENABLED
    CHECK (ENABLED IN (0, 1))
);

/* Tạo bảng history */
CREATE TABLE MACHINE_STATUS_HISTORY ( 
ID NUMBER GENERATED BY DEFAULT AS IDENTITY, 
MACHINE_ID NUMBER NOT NULL, 
STATUS NUMBER(3) NOT NULL, 
ERROR_ID NUMBER, 
START_TIME TIMESTAMP(3) NOT NULL, 
END_TIME TIMESTAMP(3), 
CONSTRAINT PK_MACHINE_STATUS_HISTORY PRIMARY KEY (ID), 
CONSTRAINT FK_MSH_MACHINE FOREIGN KEY (MACHINE_ID) REFERENCES FATP_MACHINE_INFO(MACHINE_ID), 
CONSTRAINT FK_MSH_ERROR FOREIGN KEY (ERROR_ID) REFERENCES ERROR_CATALOG(ERROR_ID), 
CONSTRAINT CK_MSH_TIME CHECK ( END_TIME IS NULL OR END_TIME >= START_TIME ) );

/* Tạo bảng constrain */

ALTER TABLE FATP_MACHINE_INFO
ADD CONSTRAINT CK_FMI_STATUS
CHECK (CURRENT_STATUS IN (0,1,2,3,4,5));

ALTER TABLE MACHINE_STATUS_HISTORY
ADD CONSTRAINT CK_MSH_STATUS
CHECK (STATUS IN (0,1,2,3,4,5));

/* Tạo index */

CREATE INDEX IX_MSH_MACHINE_TIME
ON MACHINE_STATUS_HISTORY
(
    MACHINE_ID,
    START_TIME DESC
);

/* Tạo index unique đảm bảo 1 máy có duy nhất 1 history END_TIME = NULL */
CREATE UNIQUE INDEX UX_MSH_CURRENT
ON MACHINE_STATUS_HISTORY
(
    CASE
        WHEN END_TIME IS NULL
        THEN MACHINE_ID
    END
);

/* Tạo proceduce */

Logic:

Collector gửi:
MACHINE_CODE
STATUS
ERROR_CODE
EVENT_TIME
        ↓
Khóa record máy
        ↓
Tìm ERROR_ID theo ERROR_CODE
        ↓
So sánh với trạng thái hiện tại
        │
        ├─ Không thay đổi
        │      ↓
        │   chỉ UPDATE LAST_UPDATE
        │
        └─ Có thay đổi
               ↓
        đóng END_TIME history cũ
               ↓
        INSERT history mới
               ↓
        UPDATE trạng thái hiện tại

CREATE OR REPLACE PROCEDURE UPDATE_MACHINE_STATUS
(
    P_MACHINE_CODE IN VARCHAR2,
    P_STATUS       IN NUMBER,
    P_ERROR_CODE   IN VARCHAR2 DEFAULT NULL,
    P_EVENT_TIME   IN TIMESTAMP DEFAULT NULL
)
AS
    V_EVENT_TIME          TIMESTAMP(3);

    V_MACHINE_ID          FATP_MACHINE_INFO.MACHINE_ID%TYPE;
    V_CURRENT_STATUS      FATP_MACHINE_INFO.CURRENT_STATUS%TYPE;
    V_CURRENT_ERROR_ID    FATP_MACHINE_INFO.CURRENT_ERROR_ID%TYPE;

    V_NEW_ERROR_ID        ERROR_CATALOG.ERROR_ID%TYPE;

BEGIN

    /* ================================================
       0. Thời gian event
       ================================================ */
    V_EVENT_TIME :=
        NVL(P_EVENT_TIME, CAST(SYSTIMESTAMP AS TIMESTAMP));


    /* ================================================
       1. Lấy thông tin máy + khóa record
       ================================================ */
    SELECT
        MACHINE_ID,
        CURRENT_STATUS,
        CURRENT_ERROR_ID
    INTO
        V_MACHINE_ID,
        V_CURRENT_STATUS,
        V_CURRENT_ERROR_ID
    FROM FATP_MACHINE_INFO
    WHERE MACHINE_CODE = P_MACHINE_CODE
      AND ENABLED = 1
    FOR UPDATE;


    /* ================================================
       2. Lấy ERROR_ID
       ================================================ */
    V_NEW_ERROR_ID := NULL;

    IF P_ERROR_CODE IS NOT NULL THEN

        BEGIN
            SELECT ERROR_ID
            INTO V_NEW_ERROR_ID
            FROM ERROR_CATALOG
            WHERE ERROR_CODE = P_ERROR_CODE;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                V_NEW_ERROR_ID := NULL;
        END;

    END IF;


    /* ================================================
       3. STATUS + ERROR không thay đổi
       ================================================ */
    IF NVL(V_CURRENT_STATUS, -1) = NVL(P_STATUS, -1)
       AND
       NVL(V_CURRENT_ERROR_ID, -1) = NVL(V_NEW_ERROR_ID, -1)
    THEN

        /*
           Không tạo history mới.

           Nhưng bắt buộc update LAST_UPDATE
           để chứng minh collector vẫn gửi dữ liệu.
        */
        UPDATE FATP_MACHINE_INFO
        SET LAST_UPDATE = V_EVENT_TIME
        WHERE MACHINE_ID = V_MACHINE_ID;


    ELSE

        /* ============================================
           4. Đóng history hiện tại
           ============================================ */
        UPDATE MACHINE_STATUS_HISTORY
        SET END_TIME = V_EVENT_TIME
        WHERE MACHINE_ID = V_MACHINE_ID
          AND END_TIME IS NULL;


        /* ============================================
           5. Tạo history mới
           ============================================ */
        INSERT INTO MACHINE_STATUS_HISTORY
        (
            MACHINE_ID,
            STATUS,
            ERROR_ID,
            START_TIME,
            END_TIME
        )
        VALUES
        (
            V_MACHINE_ID,
            P_STATUS,
            V_NEW_ERROR_ID,
            V_EVENT_TIME,
            NULL
        );


        /* ============================================
           6. Update trạng thái hiện tại
           ============================================ */
        UPDATE FATP_MACHINE_INFO
        SET
            CURRENT_STATUS            = P_STATUS,
            CURRENT_ERROR_ID          = V_NEW_ERROR_ID,
            START_TIME_CURRENT_STATUS = V_EVENT_TIME,
            LAST_UPDATE               = V_EVENT_TIME
        WHERE MACHINE_ID = V_MACHINE_ID;

    END IF;


EXCEPTION

    WHEN NO_DATA_FOUND THEN

        RAISE_APPLICATION_ERROR(
            -20001,
            'Machine not found or disabled: '
            || P_MACHINE_CODE
        );

    WHEN TOO_MANY_ROWS THEN

        RAISE_APPLICATION_ERROR(
            -20002,
            'Duplicate MACHINE_CODE: '
            || P_MACHINE_CODE
        );

    WHEN OTHERS THEN
        RAISE;

END;

/* Tạo Produce tự chuyển sang UNKNOWN */

CREATE OR REPLACE PROCEDURE MARK_UNKNOWN_MACHINES
(
    P_TIMEOUT_SECONDS IN NUMBER DEFAULT 10
)
AS
    V_UNKNOWN_TIME TIMESTAMP(3);
BEGIN

    FOR R IN
    (
        SELECT
            MACHINE_ID,
            CURRENT_STATUS,
            LAST_UPDATE
        FROM FATP_MACHINE_INFO
        WHERE ENABLED = 1

          -- Máy hiện tại chưa UNKNOWN
          AND CURRENT_STATUS <> 0

          -- Đã từng nhận dữ liệu
          AND LAST_UPDATE IS NOT NULL

          -- Quá timeout
          AND LAST_UPDATE <
              LOCALTIMESTAMP
              - NUMTODSINTERVAL(P_TIMEOUT_SECONDS, 'SECOND')

        FOR UPDATE SKIP LOCKED
    )
    LOOP

        /*
          Thời điểm bắt đầu UNKNOWN =
          lần cuối nhận dữ liệu + timeout
        */
        V_UNKNOWN_TIME :=
            R.LAST_UPDATE
            + NUMTODSINTERVAL(P_TIMEOUT_SECONDS, 'SECOND');


        /* 1. Đóng history hiện tại */
        UPDATE MACHINE_STATUS_HISTORY
        SET END_TIME = V_UNKNOWN_TIME
        WHERE MACHINE_ID = R.MACHINE_ID
          AND END_TIME IS NULL;


        /* 2. Tạo history UNKNOWN */
        INSERT INTO MACHINE_STATUS_HISTORY
        (
            MACHINE_ID,
            STATUS,
            ERROR_ID,
            START_TIME,
            END_TIME
        )
        VALUES
        (
            R.MACHINE_ID,
            0,                  -- UNKNOWN
            NULL,
            V_UNKNOWN_TIME,
            NULL
        );


        /* 3. Update trạng thái hiện tại */
        UPDATE FATP_MACHINE_INFO
        SET
            CURRENT_STATUS            = 0,
            CURRENT_ERROR_ID          = NULL,
            START_TIME_CURRENT_STATUS = V_UNKNOWN_TIME

            -- KHÔNG update LAST_UPDATE

        WHERE MACHINE_ID = R.MACHINE_ID;

    END LOOP;

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;

END;
/

/* Tạo schudeler chạy proceduce MARK_UNKNOWN_MACHINES với khoảng thời gian 5 giây */

BEGIN
    DBMS_SCHEDULER.CREATE_JOB
    (
        JOB_NAME        => 'JOB_CHECK_MACHINE_UNKNOWN',
        JOB_TYPE        => 'PLSQL_BLOCK',

        JOB_ACTION      => '
            BEGIN
                MARK_UNKNOWN_MACHINES(10);
            END;
        ',

        START_DATE      => SYSTIMESTAMP,

        REPEAT_INTERVAL => 'FREQ=SECONDLY;INTERVAL=5',

        ENABLED         => TRUE,
        AUTO_DROP       => FALSE,

        COMMENTS        => 'Set machine UNKNOWN when no data received for 10 seconds'
    );
END;
